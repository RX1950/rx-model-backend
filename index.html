`; feed.prepend(loaderCard); // Prepare the real payload for your Go model / backend const formData = new FormData(); formData.append('image', targetFile); formData.append('sport', selectedSport); formData.append('context', contextText); try { // Pointing to your active analytical processing endpoint const response = await fetch(`${BACKEND}/api/analyze`, { method: 'POST', body: formData }); if (!response.ok) throw new Error(`Server returned status: ${response.status}`); const data = await response.json(); // Extract your structural model writeup or fallback text if unavailable const modelWriteup = data.analysis || data.projection || JSON.stringify(data, null, 2); // Update the loader card directly with your actual real-time engine data loaderCard.innerHTML = `
